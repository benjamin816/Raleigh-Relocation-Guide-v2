function syncUnsentWebsiteLeadsToBoldTrail() {
  const ss = SpreadsheetApp.openById(getBoldTrailSpreadsheetId_());
  const summaries = [];

  const appointmentSummary = processAppointmentLeadSheet_(ss, APPOINTMENTS_LEADS_SHEET_NAME, BOLDTRAIL_MAX_ROWS_PER_RUN);
  summaries.push(appointmentSummary);

  const websiteSummary = processWebsiteLeadSheet_(ss, WEBSITE_LEADS_SHEET_NAME, BOLDTRAIL_MAX_ROWS_PER_RUN);
  summaries.push(websiteSummary);

  let totalRows = 0;
  let totalFound = 0;
  let totalProcessed = 0;
  let totalSynced = 0;
  let totalFailed = 0;

  summaries.forEach(function (summary) {
    totalRows += summary.total_rows || 0;
    totalFound += summary.found_rows || 0;
    totalProcessed += summary.pending_rows || 0;
    totalSynced += summary.synced_count || 0;
    totalFailed += summary.failed_count || 0;
    Logger.log(summary.sheet_name + ': ' + summary.found_rows + ' found, ' + summary.pending_rows + ' processed out of ' + summary.scan_limit + ' limit. Next run in ' + BOLDTRAIL_TRIGGER_MINUTES + ' minutes.');
  });

  Logger.log(
    totalFound + ' found, ' +
    totalProcessed + ' processed with a ' +
    BOLDTRAIL_MAX_ROWS_PER_RUN + '-row per-sheet limit. Next run in ' +
    BOLDTRAIL_TRIGGER_MINUTES + ' minutes.'
  );

  return {
    ok: true,
    total_rows: totalRows,
    found_rows: totalFound,
    pending_rows: totalProcessed,
    synced_count: totalSynced,
    failed_count: totalFailed,
    scan_limit_per_sheet: BOLDTRAIL_MAX_ROWS_PER_RUN,
    next_run_minutes: BOLDTRAIL_TRIGGER_MINUTES
  };
}

function syncLegacyKitLeadsToBoldTrailOnce() {
  const properties = PropertiesService.getScriptProperties();
  if (clean_(properties.getProperty(BOLDTRAIL_KIT_MIGRATION_COMPLETE_PROPERTY)).toLowerCase() === 'true') {
    return {
      ok: true,
      already_complete: true,
      message: 'The one-time Kit-to-BoldTrail migration has already completed.'
    };
  }

  const ss = SpreadsheetApp.openById(getBoldTrailSpreadsheetId_());
  const summary = processKitLeadSheet_(ss, KIT_LEADS_SHEET_NAME, BOLDTRAIL_MAX_ROWS_PER_RUN);
  const reachedRunLimit = (summary.pending_rows || 0) >= BOLDTRAIL_MAX_ROWS_PER_RUN;
  const migrationComplete = !reachedRunLimit && (summary.failed_count || 0) === 0;

  if (migrationComplete) {
    properties.setProperty(BOLDTRAIL_KIT_MIGRATION_COMPLETE_PROPERTY, 'true');
  }

  return Object.assign({}, summary, {
    ok: migrationComplete,
    migration_complete: migrationComplete,
    run_again_required: !migrationComplete,
    message: migrationComplete
      ? 'The one-time Kit-to-BoldTrail migration is complete. The scheduled trigger will not scan the Kit tab.'
      : 'The Kit migration has more rows or failures remaining. Run this function again after reviewing the failures.'
  });
}

function backfillBoldTrailLeadSourceHashtagsOnce() {
  const properties = PropertiesService.getScriptProperties();
  if (clean_(properties.getProperty(BOLDTRAIL_SOURCE_TAG_BACKFILL_COMPLETE_PROPERTY)).toLowerCase() === 'true') {
    return {
      ok: true,
      already_complete: true,
      message: 'The one-time BoldTrail lead-source hashtag backfill has already completed.'
    };
  }

  const ss = SpreadsheetApp.openById(getBoldTrailSpreadsheetId_());
  let cursors = {};
  try {
    cursors = JSON.parse(properties.getProperty(BOLDTRAIL_SOURCE_TAG_BACKFILL_CURSOR_PROPERTY) || '{}');
  } catch (_) {
    cursors = {};
  }
  const summaries = [
    backfillBoldTrailSourceTagsForSheet_(ss, KIT_LEADS_SHEET_NAME, 'kit', cursors[KIT_LEADS_SHEET_NAME] || 2, BOLDTRAIL_SOURCE_TAG_BACKFILL_ROWS_PER_SHEET),
    backfillBoldTrailSourceTagsForSheet_(ss, WEBSITE_LEADS_SHEET_NAME, 'website', cursors[WEBSITE_LEADS_SHEET_NAME] || 2, BOLDTRAIL_SOURCE_TAG_BACKFILL_ROWS_PER_SHEET),
    backfillBoldTrailSourceTagsForSheet_(ss, APPOINTMENTS_LEADS_SHEET_NAME, 'appointment', cursors[APPOINTMENTS_LEADS_SHEET_NAME] || 2, BOLDTRAIL_SOURCE_TAG_BACKFILL_ROWS_PER_SHEET)
  ];
  const failedCount = summaries.reduce(function (total, summary) {
    return total + (summary.failed_count || 0);
  }, 0);
  const hasMore = summaries.some(function (summary) { return Boolean(summary.has_more); });
  const complete = failedCount === 0 && !hasMore;

  summaries.forEach(function (summary) {
    if (summary.has_more) {
      cursors[summary.sheet_name] = summary.next_row;
    } else {
      delete cursors[summary.sheet_name];
    }
  });

  if (complete) {
    properties.setProperty(BOLDTRAIL_SOURCE_TAG_BACKFILL_COMPLETE_PROPERTY, 'true');
    properties.deleteProperty(BOLDTRAIL_SOURCE_TAG_BACKFILL_CURSOR_PROPERTY);
  } else {
    properties.setProperty(BOLDTRAIL_SOURCE_TAG_BACKFILL_CURSOR_PROPERTY, JSON.stringify(cursors));
  }

  return {
    ok: complete,
    backfill_complete: complete,
    run_again_required: !complete,
    failed_count: failedCount,
    summaries: summaries,
    message: complete
      ? 'The one-time BoldTrail lead-source hashtag backfill is complete.'
      : 'The source-tag backfill has more rows or failures remaining. Review the summary and run it again.'
  };
}

function backfillBoldTrailSourceTagsForSheet_(ss, sheetName, sheetType, startRow, scanLimit) {
  const sheet = requireSheetByName_(ss, sheetName);
  const indexes = headerIndexes_(sheet);
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  const limit = Math.max(0, Math.min(Number(scanLimit) || 0, BOLDTRAIL_MAX_ROWS_PER_RUN));
  const availableRows = Math.max(0, lastRow - 1);
  let scanned = 0;
  let processed = 0;
  let synced = 0;
  let failed = 0;
  let skipped = 0;
  let retryRow = 0;

  let rowNumber = Math.max(2, Number(startRow) || 2);
  for (; rowNumber <= lastRow && scanned < limit; rowNumber++) {
    scanned += 1;
    const row = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];
    let context = null;
    if (sheetType === 'kit') {
      context = kitLeadContextFromSheetRow_(row, indexes);
    } else if (sheetType === 'appointment') {
      context = appointmentLeadContextFromSheetRow_(row, indexes);
    } else {
      context = leadContextFromSheetRow_(row, indexes);
    }

    const lead = context && context.lead;
    if (!lead || !hasText_(lead.email)) {
      skipped += 1;
      continue;
    }

    const isReloGuideLead = sheetType === 'kit' || isReloGuideLead_(lead);
    const tagName = isReloGuideLead
      ? BOLDTRAIL_RELO_TAG
      : (sheetType === 'appointment'
        ? BOLDTRAIL_NEWSLETTER_TAG
        : (leadHasPhoneNumber_(lead) ? BOLDTRAIL_NEWSLETTER_TAG : BOLDTRAIL_NO_PHONE_TAG));
    const result = syncLeadToBoldTrail_(lead, tagName, {
      forceTag: isReloGuideLead,
      preferEmailLookup: true
    });
    processed += 1;

    if (result.status === 'synced') {
      synced += 1;
    } else {
      failed += 1;
      retryRow = rowNumber;
      break;
    }
  }

  return {
    sheet_name: sheetName,
    total_rows: availableRows,
    scanned_rows: scanned,
    processed_rows: processed,
    synced_count: synced,
    failed_count: failed,
    skipped_count: skipped,
    scan_limit: limit,
    next_row: retryRow || rowNumber,
    has_more: Boolean(retryRow) || rowNumber <= lastRow
  };
}

function syncLeadToBoldTrail_(lead, tagName, options) {
  try {
    if (!lead || typeof lead !== 'object') {
      return { status: 'skipped_invalid_lead', error: 'Lead payload was not a valid object.' };
    }

    const email = clean_(lead.email || lead.home_eval_email);
    const phoneDigits = phoneFromFullPhone_(lead.phone || lead.home_eval_phone);
    if (!email) {
      return { status: 'skipped_no_email', error: 'No email address provided for BoldTrail sync.' };
    }

    const apiToken = getBoldTrailApiKey_();
    if (!apiToken) {
      return { status: 'skipped_no_api_key', error: 'BOLDTRAIL_API_KEY is not configured in Script properties.' };
    }

    let existingContacts = [];
    if (options && options.preferEmailLookup) {
      existingContacts = boldTrailFindContactsByEmail_(apiToken, email);
      if (!existingContacts.length && phoneDigits) {
        existingContacts = boldTrailFindContactsByPhone_(apiToken, phoneDigits);
      }
    } else {
      existingContacts = boldTrailFindMatchingContacts_(apiToken, email, phoneDigits);
    }
    const existingContact = existingContacts.length ? existingContacts[0] : null;
    const existingHasPhone = existingContacts.some(function (contact) {
      return boldTrailContactHasPhone_(contact);
    });
    const incomingHasPhone = leadHasPhoneNumber_(lead);
    const hasPhone = incomingHasPhone || existingHasPhone;
    const forceTag = Boolean(options && options.forceTag);
    const resolvedTag = forceTag ? clean_(tagName) : (clean_(tagName) || boldTrailTagForLead_(lead));
    const resolvedTags = boldTrailUniqueTagNames_([resolvedTag].concat(boldTrailSourceTagsForLead_(lead)));
    const finalStatus = hasPhone ? BOLDTRAIL_PROSPECT_STATUS : BOLDTRAIL_LEAD_STATUS;
    const contactPayload = boldTrailContactPayload_(lead, existingContact, hasPhone, finalStatus, incomingHasPhone);
    const targetContacts = existingContacts.slice();
    let createdContact = null;

    if (!targetContacts.length) {
      createdContact = boldTrailCreateContact_(apiToken, contactPayload);
      delete boldTrailContactCache_[email.toLowerCase()];
      if (phoneDigits) {
        delete boldTrailContactPhoneCache_[phoneDigits];
      }
      targetContacts.push.apply(targetContacts, boldTrailFindMatchingContacts_(apiToken, email, phoneDigits));
    }

    if (!targetContacts.length && createdContact && createdContact.id) {
      targetContacts.push(createdContact);
    }

    const uniqueContacts = boldTrailUniqueContacts_(targetContacts);
    if (!uniqueContacts.length) {
      throw new Error('BoldTrail contact ID could not be resolved for ' + email + '.');
    }

    const contactStatePayload = {
      status: finalStatus,
      phone_on: hasPhone ? 1 : 0,
      text_on: hasPhone ? 1 : 0,
      email_optin: 1
    };
    if (incomingHasPhone) {
      contactStatePayload.tcpa_optin_date = boldTrailNowUtcString_();
    }

    uniqueContacts.forEach(function (contact) {
      const contactId = String(contact.id || '');
      if (!contactId) {
        return;
      }
      boldTrailUpdateContact_(apiToken, contactId, Object.assign({}, contactPayload, contactStatePayload));
      boldTrailEnsureContactStatus_(apiToken, contactId, finalStatus, contactPayload, contactStatePayload);
      resolvedTags.forEach(function (resolvedTagName) {
        boldTrailAddHashtag_(apiToken, contactId, resolvedTagName);
      });
    });

    return {
      status: 'synced',
      error: '',
      route: 'boldtrail',
      contact_id: String(uniqueContacts[0].id || ''),
      tags: resolvedTags
    };
  } catch (err) {
    return {
      status: 'failed',
      error: clean_(err && err.message ? err.message : err),
      route: '',
      contact_id: ''
    };
  }
}

function processKitLeadSheet_(ss, sheetName, scanLimit) {
  const sheet = requireSheetByName_(ss, sheetName);
  ensureLeadSheetBoldTrailHeaders_(sheet);

  const indexes = headerIndexes_(sheet);
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  const limit = Math.max(0, Math.min(Number(scanLimit) || 0, BOLDTRAIL_MAX_ROWS_PER_RUN));

  let found = 0;
  let processed = 0;
  let synced = 0;
  let failed = 0;

  for (let rowNumber = 2; rowNumber <= lastRow && processed < limit; rowNumber++) {
    const row = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];
    const syncStatus = clean_(row[indexes.boldtrail_sync_status]).toLowerCase();
    if (syncStatus === 'synced' || syncStatus === 'skipped_no_email') {
      continue;
    }

    const context = kitLeadContextFromSheetRow_(row, indexes);
    const lead = context.lead;
    if (!lead) {
      continue;
    }

    if (!hasText_(lead.email)) {
      updateLeadSheetBoldTrailStatus_(sheet, rowNumber, 'skipped_no_email', '', '', 'No email address provided for BoldTrail sync.');
      continue;
    }

    if (!isEligibleForBoldTrail_(lead)) {
      if (syncStatus !== 'waiting_1_day') {
        updateLeadSheetBoldTrailStatus_(sheet, rowNumber, 'waiting_1_day', '', '', 'Waiting for 1-day hold to expire before syncing to BoldTrail.');
      }
      continue;
    }

    found += 1;
    const result = syncLeadToBoldTrail_(lead, BOLDTRAIL_RELO_TAG, { forceTag: true });
    processed += 1;

    if (result.status === 'synced') {
      synced += 1;
      updateLeadSheetBoldTrailStatus_(sheet, rowNumber, 'synced', result.contact_id || '', new Date(), '');
    } else {
      failed += 1;
      updateLeadSheetBoldTrailStatus_(sheet, rowNumber, 'failed', '', '', result.error || 'Unknown BoldTrail sync error.');
    }
  }

  return {
    sheet_name: sheetName,
    total_rows: Math.max(0, lastRow - 1),
    found_rows: found,
    pending_rows: processed,
    synced_count: synced,
    failed_count: failed,
    scan_limit: limit
  };
}

function isEligibleForBoldTrail_(lead) {
  if (!lead || typeof lead !== 'object') {
    return false;
  }
  if (BOLDTRAIL_MIN_AGE_DAYS <= 0) {
    return true;
  }
  const submittedAt = new Date(lead.submitted_at || '');
  if (isNaN(submittedAt.getTime())) {
    return false;
  }
  const minAgeMs = BOLDTRAIL_MIN_AGE_DAYS * 24 * 60 * 60 * 1000;
  return (Date.now() - submittedAt.getTime()) >= minAgeMs;
}

function kitLeadContextFromSheetRow_(row, indexes) {
  const lead = normalizeLead_({
    submitted_at: row[indexes.timestamp],
    source_page: KIT_LEADS_SHEET_NAME,
    source_key: 'kit',
    lead_type: row[indexes.lead_type] || 'relo_guide',

  return { lead: lead, raw: {} };
}

function appointmentLeadContextFromSheetRow_(row, indexes) {
  const requestedIntent = appointmentRequestedContactIntent_(row, indexes);
  const lead = normalizeLead_({
    submitted_at: row[indexes.logged_at] || row[indexes.scheduled_at] || row[indexes.event_start] || row[indexes.send_at],
    source_page: APPOINTMENTS_LEADS_SHEET_NAME,
    source_key: 'appointment',
    lead_type: 'requested_contact',
    requested_contact_source: 'direct_booking',
    requested_contact_intent: requestedIntent,
    first_name: row[indexes.first_name],
    last_name: row[indexes.last_name],
    email: row[indexes.email],
    phone: row[indexes.phone],
    submission_id: row[indexes.event_id]
  });

  return { lead: lead, raw: {} };
}

function appointmentRequestedContactIntent_(row, indexes) {
  const haystack = [
    row[indexes.title],
    row[indexes.preference_note],
    row[indexes.pre_call_notes]
  ].map(clean_).join(' ').toLowerCase();

  if (!haystack) {
    return '';
  }
  if (/\b(sell|seller|listing|list)\b/.test(haystack)) {
    return 'looking_to_sell';
  }
  if (/\b(buy|buyer|purchase|home search)\b/.test(haystack)) {
    return 'looking_to_buy';
  }
  return '';
}

function ensureHeadersPresent_(sheet, requiredHeaders) {
  const lastCol = Math.max(1, sheet.getLastColumn());
  const headerRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  const normalized = headerRow.map(function (h) { return clean_(h).toLowerCase(); });

  requiredHeaders.forEach(function (headerName) {
    const key = clean_(headerName).toLowerCase();
    if (normalized.indexOf(key) !== -1) {
      return;
    }

    const blankIndex = normalized.indexOf('');
    if (blankIndex !== -1) {
      sheet.getRange(1, blankIndex + 1).setValue(headerName);
      normalized[blankIndex] = key;
      return;
    }

    const appendAt = sheet.getLastColumn() + 1;
    sheet.getRange(1, appendAt).setValue(headerName);
    normalized.push(key);
  });

  if (sheet.getFrozenRows() !== 1) {
    sheet.setFrozenRows(1);
  }
}

function ensureLeadSheetBoldTrailHeaders_(sheet) {
  ensureHeadersPresent_(sheet, [
    'BoldTrail Sync Status',
    'BoldTrail Contact ID',
    'BoldTrail Synced At',
    'BoldTrail Sync Error'
  ]);
}

function boldTrailDealType_(lead) {

function boldTrailSourceTagsForLead_(lead) {
  if (!lead || typeof lead !== 'object') {
    return [];
  }

  const tags = [];
  const leadType = clean_(lead.lead_type).toLowerCase();
  const requestedSource = clean_(lead.requested_contact_source).toLowerCase();
  const sourcePage = clean_(lead.source_page);
  const sourceKey = clean_(lead.source_key);
  const sourceText = [sourcePage, sourceKey].join(' ').toLowerCase();
  const sourcePaths = [sourcePage, sourceKey].map(boldTrailSourcePath_).filter(Boolean);

  if (isReloGuideLead_(lead)) {
    tags.push('lead_source_relocation_guide');
  }

  if (
    leadType === 'requested_contact' && requestedSource.indexOf('home_call_me_popup') !== -1
  ) {
    tags.push('lead_source_call_me_popup');
  }

  if (leadType === 'suburb_quiz') {
    tags.push('lead_source_suburb_quiz');
  }

  if (leadType === 'neighborhood_quiz' || clean_(lead.quiz_type).toLowerCase() === 'neighborhood_fit') {
    tags.push('lead_source_neighborhood_quiz');
    const quizCitySlug = boldTrailTagSlug_(lead.quiz_city);
    if (quizCitySlug) {
      tags.push('lead_source_' + quizCitySlug + '_neighborhood_quiz');
    }
  }

  if (leadType === 'new_construction' || requestedSource.indexOf('new_construction') !== -1) {
    tags.push('lead_source_new_construction');
  }

  if (leadType === 'home_eval') {
    tags.push('lead_source_home_valuation');
  }

  if (
    leadType === 'requested_contact' &&
    (clean_(lead.source_key).toLowerCase() === 'appointment' || sourceText.indexOf('appointments booked directly') !== -1)
  ) {
    tags.push('lead_source_consult_booking');
  }

  if (sourcePaths.some(function (path) { return path === '/buy' || path.indexOf('/buy/') === 0; })) {
    tags.push('lead_source_buyer_page');
  }
  if (sourcePaths.some(function (path) { return path === '/sell' || path.indexOf('/sell/') === 0; })) {
    tags.push('lead_source_seller_page');
  }
  if (sourcePaths.some(function (path) { return path === '/consult' || path.indexOf('/consult/') === 0; })) {
    tags.push('lead_source_consult_page');
  }
  if (sourcePaths.some(function (path) { return path === '/relocation' || path.indexOf('/relocation/') === 0; })) {
    tags.push('lead_source_relocation_page');
  }
  if (sourcePaths.some(function (path) { return path === '/'; })) {
    tags.push('lead_source_homepage');
  }

  if (!tags.length) {
    tags.push('lead_source_website_form');
  }

  return boldTrailUniqueTagNames_(tags);
}

function boldTrailSourcePath_(value) {
  const normalized = clean_(value).toLowerCase();
  if (!normalized) {
    return '';
  }
  let path = normalized.replace(/^https?:\/\/[^/]+/i, '');
  path = path.split('#')[0].split('?')[0];
  if (!path) {
    return '/';
  }
  if (path.charAt(0) !== '/') {
    return '';
  }
  return path.replace(/\/+$/, '') || '/';
}

function boldTrailTagSlug_(value) {
  return clean_(value)
    .toLowerCase()
    .replace(/&/g, ' and ')
    .replace(/[^a-z0-9]+/g, '_')
    .replace(/^_+|_+$/g, '');
}

function boldTrailUniqueTagNames_(tagNames) {
  const seen = {};
  return (Array.isArray(tagNames) ? tagNames : []).map(function (tagName) {
    return boldTrailTagSlug_(tagName);
  }).filter(function (tagName) {
    if (!tagName || seen[tagName]) {
      return false;
    }
    seen[tagName] = true;
    return true;
  });
}

function boldTrailFindContactByEmail_(apiToken, email) {

function boldTrailFindContactsByEmail_(apiToken, email) {
  const normalizedEmail = clean_(email);
  if (!normalizedEmail) {
    return [];
  }

  const cacheKey = normalizedEmail.toLowerCase();
  if (boldTrailContactCache_[cacheKey]) {
    return [boldTrailContactCache_[cacheKey]];
  }

  const response = boldTrailRequest_(
    apiToken,
    BOLDTRAIL_API_BASE + '/contacts?filter[email]=' + encodeURIComponent(normalizedEmail) + '&limit=10&includeArchived=1',
    'get'
  );
  const contacts = Array.isArray(response) ? response : (response && response.data) ? response.data : [];
  const matches = contacts.filter(function (contact) {
    return clean_(contact && contact.email).toLowerCase() === cacheKey;
  });
  if (matches.length) {
    boldTrailContactCache_[cacheKey] = matches[0];
  }
  return matches;
}

function boldTrailFindContactByPhone_(apiToken, phoneDigits) {
  const contacts = boldTrailFindContactsByPhone_(apiToken, phoneDigits);
  return contacts.length ? contacts[0] : null;
}

function boldTrailFindContactsByPhone_(apiToken, phoneDigits) {
  const normalizedPhone = phoneFromFullPhone_(phoneDigits);
  if (!normalizedPhone) {
    return [];
  }

  const cacheKey = normalizedPhone;
  if (boldTrailContactPhoneCache_[cacheKey]) {
    return [boldTrailContactPhoneCache_[cacheKey]];
  }

  const matches = [];
  const pageSize = 500;
  for (let page = 1; page <= 20; page++) {
    const response = boldTrailRequest_(
      apiToken,
      BOLDTRAIL_API_BASE + '/contacts?limit=' + pageSize + '&page=' + page + '&includeArchived=1',
      'get'
    );
    const contacts = Array.isArray(response) ? response : (response && response.data) ? response.data : [];
    if (!contacts.length) {
      break;
    }

    const pageMatches = contacts.filter(function (contact) {
      const contactPhones = [
        contact && contact.phone,
        contact && contact.cell_phone_1,
        contact && contact.cell_phone,
        contact && contact.mobile_phone,
        contact && contact.home_phone
      ].map(phoneFromFullPhone_).filter(Boolean);
      return contactPhones.indexOf(normalizedPhone) !== -1;
    });

    if (pageMatches.length) {
      matches.push.apply(matches, pageMatches);
    }

    if (contacts.length < pageSize) {
      break;
    }
  }

  if (matches.length) {
    boldTrailContactPhoneCache_[cacheKey] = matches[0];
  }
  return matches;
}

function boldTrailFindMatchingContacts_(apiToken, email, phoneDigits) {
  const emailMatches = boldTrailFindContactsByEmail_(apiToken, email);
  const phoneMatches = phoneDigits ? boldTrailFindContactsByPhone_(apiToken, phoneDigits) : [];
  return boldTrailUniqueContacts_(emailMatches.concat(phoneMatches));
}

    seen[id] = true;
    return true;
  });
}

function boldTrailEnsureContactStatus_(apiToken, contactId, expectedStatus, contactPayload, contactStatePayload) {
  const statusValue = Number(expectedStatus);
  if (!hasText_(contactId) || !Number.isFinite(statusValue)) {
    return {};
  }

  const delays = [0, 1000, 2000];
  let lastContact = {};
  for (let i = 0; i < delays.length; i++) {
    if (delays[i]) {
      Utilities.sleep(delays[i]);
    }
    lastContact = boldTrailGetContact_(apiToken, contactId) || {};
    if (Number(lastContact && lastContact.status) === statusValue) {
      return { ok: true };
    }
    boldTrailRequest_(apiToken, BOLDTRAIL_API_BASE + '/contact/' + encodeURIComponent(contactId), 'put', Object.assign({}, contactPayload, contactStatePayload, { status: statusValue }));
  }

  throw new Error('BoldTrail contact status ' + statusValue + ' was not visible on contact ' + contactId + '. Current contact: ' + JSON.stringify(lastContact));
}

function boldTrailAddHashtag_(apiToken, contactId, tagName) {
  if (!hasText_(contactId) || !hasText_(tagName)) {
    return {};
  }
  const normalizedTag = clean_(tagName);
  const baseUrl = BOLDTRAIL_API_BASE + '/contact/' + encodeURIComponent(contactId) + '/tags';
  const attempts = [
    { tags: [normalizedTag] },
    { tags: [{ name: normalizedTag, locked: 0 }] }
  ];
  let lastResponse = {};
  let lastError = '';

  for (let i = 0; i < attempts.length; i++) {
    try {
      lastResponse = boldTrailRequest_(apiToken, baseUrl, 'put', attempts[i]);
      if (boldTrailTagExists_(lastResponse, normalizedTag)) {
        return { ok: true };
      }
      if (boldTrailTagExists_(boldTrailGetContactTags_(apiToken, contactId), normalizedTag)) {
        return { ok: true };
      }
    } catch (err) {
      lastError = clean_(err && err.message ? err.message : err);
    }
  }
  return boldTrailEnsureContactTag_(apiToken, contactId, normalizedTag, lastError);
}

  if (!hasText_(contactId)) {
    return [];
  }
  const response = boldTrailRequest_(apiToken, BOLDTRAIL_API_BASE + '/contact/' + encodeURIComponent(contactId) + '/tags', 'get');
  if (Array.isArray(response)) {
    return response;
  }
  if (response && Array.isArray(response.tags)) {
    return response.tags;
  }
  if (response && Array.isArray(response.data)) {
    return response.data;
  }
  return [];
}

function boldTrailEnsureContactTag_(apiToken, contactId, tagName, priorError) {
  const delays = [0, 1000, 2000];
  let lastTags = [];

  for (let i = 0; i < delays.length; i++) {
    if (delays[i]) {
      Utilities.sleep(delays[i]);
    }
    const currentTags = boldTrailGetContactTags_(apiToken, contactId);
    lastTags = currentTags;
    if (boldTrailTagExists_(currentTags, tagName)) {
      return { ok: true };
    }
  }

  throw new Error((priorError ? priorError + ' ' : '') + 'BoldTrail tag "' + tagName + '" was not visible on contact ' + contactId + '. Current tags: ' + JSON.stringify(lastTags));
}

function boldTrailTagExists_(tags, tagName) {
  const target = clean_(tagName).toLowerCase();
  if (!target) {
    return false;
  }
  const collection = Array.isArray(tags)
    ? tags
    : (tags && Array.isArray(tags.tags))
      ? tags.tags
      : (tags && tags.data && Array.isArray(tags.data.tags))
        ? tags.data.tags
      : (tags && Array.isArray(tags.data))
        ? tags.data
        : [];
  return collection.some(function (tag) {
    if (typeof tag === 'string') {
      return clean_(tag).toLowerCase() === target;
    }
    return clean_(tag && (tag.name || tag.tag || tag.tag_name || tag.hashtag || tag.label || tag.slug || tag.value)).toLowerCase() === target;
  });
}

function boldTrailRequest_(apiToken, url, method, payload) {
  const options = {
    method: String(method || 'get').toLowerCase(),
    muteHttpExceptions: true,
    headers: {
      Authorization: 'Bearer ' + clean_(apiToken),
      Accept: 'application/json'
    }
  };

  if (payload && Object.keys(payload).length) {
    options.contentType = 'application/json';
    options.payload = JSON.stringify(payload);
  }

  const response = UrlFetchApp.fetch(url, options);
  const code = response.getResponseCode();
  const bodyText = response.getContentText() || '';
  if (code < 200 || code >= 300) {
    throw new Error('BoldTrail API ' + options.method.toUpperCase() + ' ' + url + ' failed (' + code + '): ' + bodyText);
  }
  if (!bodyText) {
    return {};
  }
  try {
    return JSON.parse(bodyText);
  } catch (_) {
    return {};
  }
}

function ensureBoldTrailHeaders_(sheet) {
  const required = headers_();
  const lastCol = Math.max(1, sheet.getLastColumn());
  const headerRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  const normalized = headerRow.map(function (h) { return clean_(h).toLowerCase(); });

  required.forEach(function (headerName) {
    const key = clean_(headerName).toLowerCase();
    if (normalized.indexOf(key) !== -1) {
      return;
    }

    const blankIndex = normalized.indexOf('');
    if (blankIndex !== -1) {
      sheet.getRange(1, blankIndex + 1).setValue(headerName);
      normalized[blankIndex] = key;
      return;
    }

    const appendAt = sheet.getLastColumn() + 1;
    sheet.getRange(1, appendAt).setValue(headerName);
    normalized.push(key);
  });

  if (sheet.getFrozenRows() !== 1) {
    sheet.setFrozenRows(1);
  }
}

  ) {
    return;
  }

  let syncedAtValue = '';
  if (syncedAt instanceof Date && !isNaN(syncedAt.getTime())) {
    syncedAtValue = syncedAt;
  }

  sheet
    .getRange(rowNumber, indexes.boldtrail_sync_status + 1, 1, 4)
    .setValues([[clean_(status), clean_(contactId), syncedAtValue, clean_(errorMessage)]]);
}

function leadHasPhoneNumber_(lead) {
  if (!lead || typeof lead !== 'object') {
    return false;
  }
  return hasText_(lead.phone) || hasText_(lead.home_eval_phone);
}

function boldTrailNowUtcString_() {
  return Utilities.formatDate(new Date(), 'UTC', 'yyyy-MM-dd HH:mm:ss');
}

function isReloGuideLead_(lead) {
  if (!lead || typeof lead !== 'object') {
    return false;
  }
  const haystack = [
    lead.source_page,
    lead.source_key,
    lead.lead_type,
    lead.requested_contact_source,
    lead.submission_id
  ].map(clean_).join(' ').toLowerCase();
  return haystack.indexOf('relo') !== -1 || haystack.indexOf('relocation guide') !== -1 || haystack.indexOf('kit') !== -1;
}

function phoneFromFullPhone_(value) {
  const digits = String(value == null ? '' : value).replace(/[^\d]/g, '');
  if (!digits) {
    return '';
  }
  if (digits.length > 10) {
    return digits.slice(-10);
  }
  return digits;
}

function normalizeRequestedContactIntent_(value) {
  const normalized = clean_(value).toLowerCase();
  if (!normalized) {
    return '';
  }
  if (
    normalized === 'looking to buy' ||
    normalized === 'looking_to_buy' ||
    normalized === 'buyer'
  ) {
    return 'looking_to_buy';
  }
  if (
    normalized === 'looking to sell' ||
    normalized === 'looking_to_sell' ||
    normalized === 'seller'
  ) {
    return 'looking_to_sell';
  }
  if (
    normalized === 'looking to sell and purchase a home' ||
    normalized === 'looking_to_sell_and_buy' ||
    normalized === 'looking to sell and buy' ||
    normalized === 'seller_and_buyer' ||
    normalized === 'sell_and_buy'
  ) {
    return 'looking_to_sell_and_buy';
  }
  return normalized.replace(/\s+/g, '_');
}

function clean_(v) {
  return String(v == null ? '' : v).replace(/\s+/g, ' ').trim();
}

function hasText_(value) {
  return clean_(value) !== '';
}

function toBoolean_(value, fallback) {
  if (typeof value === 'boolean') {
    return value;
  }
  const normalized = clean_(value).toLowerCase();
  if (!normalized) {
    return Boolean(fallback);
  }
  return normalized === 'true' || normalized === '1' || normalized === 'yes' || normalized === 'y';
}

function findRawJsonInRow_(row) {
  for (var i = 0; i < row.length; i++) {
    var candidate = clean_(row[i]);
    if (!candidate || candidate.charAt(0) !== '{' || candidate.charAt(candidate.length - 1) !== '}') {
      continue;
    }
    try {
      var parsed = JSON.parse(candidate);
      if (parsed && typeof parsed === 'object' && (parsed.submitted_at || parsed.source_page || parsed.email || parsed.lead_type)) {
        return candidate;
      }
    } catch (_) {}
  }
  return '';
}

function toIsoString_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return value.toISOString();
  }
  const parsed = new Date(value);
  if (!isNaN(parsed.getTime())) {
    return parsed.toISOString();
  }
  return new Date().toISOString();
}

function resetBoldTrailLeadSyncState() {
  const ss = SpreadsheetApp.openById(getBoldTrailSpreadsheetId_());
  const sheetNames = [
    WEBSITE_LEADS_SHEET_NAME,
    APPOINTMENTS_LEADS_SHEET_NAME
  ];

  let clearedSheets = 0;
  let clearedCells = 0;

  sheetNames.forEach(function (sheetName) {
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      return;
    }

    let indexes = null;
    if (sheetName === WEBSITE_LEADS_SHEET_NAME) {
      ensureBoldTrailHeaders_(sheet);
    } else {
      ensureLeadSheetBoldTrailHeaders_(sheet);
    }

    indexes = headerIndexes_(sheet);
    const columns = [];
    if (indexes.boldtrail_status >= 0) columns.push(indexes.boldtrail_status + 1);
    if (indexes.boldtrail_sent_at >= 0) columns.push(indexes.boldtrail_sent_at + 1);
    if (indexes.boldtrail_error >= 0) columns.push(indexes.boldtrail_error + 1);
    if (indexes.boldtrail_sync_status >= 0) columns.push(indexes.boldtrail_sync_status + 1);
    if (indexes.boldtrail_contact_id >= 0) columns.push(indexes.boldtrail_contact_id + 1);
    if (indexes.boldtrail_synced_at >= 0) columns.push(indexes.boldtrail_synced_at + 1);
    if (indexes.boldtrail_sync_error >= 0) columns.push(indexes.boldtrail_sync_error + 1);

    if (!columns.length) {
      return;
    }

    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      clearedSheets += 1;
      return;
    }

    columns.forEach(function (columnNumber) {
      sheet.getRange(2, columnNumber, lastRow - 1, 1).clearContent();
      clearedCells += lastRow - 1;
    });
    clearedSheets += 1;
  });

  return 'Cleared BoldTrail sync state on ' + clearedSheets + ' sheet(s) across ' + clearedCells + ' cell(s).';
}

function resetLegacyKitBoldTrailMigration() {
  const ss = SpreadsheetApp.openById(getBoldTrailSpreadsheetId_());
  const sheet = requireSheetByName_(ss, KIT_LEADS_SHEET_NAME);
  ensureLeadSheetBoldTrailHeaders_(sheet);
  const indexes = headerIndexes_(sheet);
  const columns = [
    indexes.boldtrail_sync_status,
    indexes.boldtrail_contact_id,
    indexes.boldtrail_synced_at,
    indexes.boldtrail_sync_error
  ].filter(function (index) { return index >= 0; });
  const lastRow = sheet.getLastRow();

  if (lastRow > 1) {
    columns.forEach(function (columnIndex) {
      sheet.getRange(2, columnIndex + 1, lastRow - 1, 1).clearContent();
    });
  }

  PropertiesService.getScriptProperties().deleteProperty(BOLDTRAIL_KIT_MIGRATION_COMPLETE_PROPERTY);
  return 'Reset the legacy Kit migration state. Run syncLegacyKitLeadsToBoldTrailOnce() to migrate it again.';
}
