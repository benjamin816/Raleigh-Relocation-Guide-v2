function sendPendingSuburbQuizEmails() {
  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  const sheet = getOrCreateSheet_(ss, CONFIG.SHEET_NAME);
  ensureQuizEmailColumns_(sheet);

  const indexes = headerIndexes_(sheet);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    return { ok: true, scanned: 0, matched: 0, sent: 0, failed: 0 };
  }

  const values = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues();
  let matched = 0;
  let sent = 0;
  let failed = 0;

  values.forEach(function (row, idx) {
    const rowNumber = idx + 2;
    const sourcePage = clean_(row[indexes.source_page]);
    const sourceKey = clean_(row[indexes.source_key]);
    const email = clean_(row[indexes.email]);
    const status = clean_(row[indexes.quiz_email_status]).toLowerCase();
    const quizType = clean_(row[indexes.quiz_type]);
    const quizTitle = clean_(row[indexes.quiz_title]);
    const quizResultsJson = clean_(row[indexes.quiz_results_json]);

    const sourceHaystack = (sourcePage + ' ' + sourceKey).toLowerCase();
    const isQuizSource = sourceHaystack.indexOf(CONFIG.SOURCE_MATCH) !== -1 ||
      sourceHaystack.indexOf('raleigh-suburb-quiz') !== -1 ||
      Boolean(quizType) ||
      Boolean(quizTitle) ||
      Boolean(quizResultsJson);
    if (!isQuizSource || !email || status === 'sent') {
      return;
    }

    matched += 1;
    try {
      const payload = rowToPayload_(row, indexes);
      if (isNeighborhoodQuizPayload_(payload)) {
        sendNeighborhoodResultsEmail_(payload);
      } else {
        sendSuburbResultsEmail_(payload);
      }
      markQuizEmailStatus_(sheet, rowNumber, 'sent', new Date(), '');
      sent += 1;
    } catch (err) {
      const message = clean_(err && err.message ? err.message : err);
      markQuizEmailStatus_(sheet, rowNumber, 'failed', '', message);
      failed += 1;
    }
  });

  return { ok: true, scanned: values.length, matched: matched, sent: sent, failed: failed };
}

function installSuburbQuizEmailTrigger() {
  removeSuburbQuizEmailTriggers();
  ScriptApp.newTrigger('sendPendingSuburbQuizEmails')
    .timeBased()
    .everyMinutes(10)
    .create();
  return { ok: true, message: 'Trigger installed: sendPendingSuburbQuizEmails every 10 minutes.' };
}

function removeSuburbQuizEmailTriggers() {
  const triggers = ScriptApp.getProjectTriggers();
  let removed = 0;
  triggers.forEach(function (trigger) {
    if (trigger.getHandlerFunction() === 'sendPendingSuburbQuizEmails') {
      ScriptApp.deleteTrigger(trigger);
      removed += 1;
    }
  });
  return { ok: true, removed: removed };
}

function sendSuburbResultsEmail_(lead, isTest) {
  const toEmail = clean_(lead.email);
  if (!toEmail) throw new Error('Missing lead email.');

  const topPick = clean_(lead.quiz_top_pick);
  const secondPick = clean_(lead.quiz_second_pick);
  const thirdPick = clean_(lead.quiz_third_pick);
  const dynamicResults = parseQuizResultsPayload_(lead.quiz_results_json);
  const useDynamicResults = dynamicResults.length > 0;
  const firstName = clean_(lead.first_name) || 'there';
  const quizTitle = clean_(lead.quiz_title) || 'Raleigh Suburb Quiz';
  const quizCity = clean_(lead.quiz_city) || 'Raleigh';
  const subject = 'Your Raleigh Suburb Quiz Results Are In!';

  if (!useDynamicResults && !topPick) throw new Error('Missing quiz top pick.');

  const fallbackPicks = [topPick, secondPick, thirdPick].filter(function (v, i, arr) {
    return v && arr.indexOf(v) === i;
  });
  const suburbCards = useDynamicResults
    ? dynamicResults.map(function (result, idx) {
        return buildNeighborhoodFitBlockHtml_(result, idx, quizCity);
      }).join('')
    : fallbackPicks.map(function (suburbName, idx) {
        return buildSuburbBlockHtml_(suburbName, idx);
      }).join('');
  const textBody = useDynamicTextBody_(firstName, useDynamicResults ? dynamicResults : fallbackPicks);

  const htmlBody =
    '<div style="font-family:Arial,sans-serif;color:#1F2937;line-height:1.55;max-width:760px;margin:0 auto;padding:20px;">' +
      '<div style="text-align:center;margin:0 0 10px;"><a href="' + CONFIG.BASE_SITE_URL + '/" target="_blank" rel="noopener noreferrer" style="display:inline-block;text-decoration:none;"><img src="' + escapeHtml_(CONFIG.LOGO_URL) + '" alt="Living in Raleigh Team" style="max-width:180px;width:100%;height:auto;display:block;margin:0 auto;"></a></div>' +
      '<h1 style="text-align:center;color:#2F5233;margin:0 0 8px;">' + escapeHtml_(quizTitle) + '</h1>' +
      '<p style="text-align:center;margin:0 0 24px;">Hi ' + escapeHtml_(firstName) + ', here are your personalized neighborhood matches based on your quiz responses.</p>' +
      suburbCards +
      '<div style="margin-top:24px;padding:16px;border:1px solid #d9e2e6;border-radius:12px;background:#f7fbf8;text-align:center;">' +
        '<h3 style="margin:0 0 8px;color:#2F5233;">Want help choosing the best fit?</h3>' +
        '<p style="margin:0 0 10px;">We can help you compare commute, schools, home styles, and budget in real time.</p>' +
        '<a href="' + CONFIG.BASE_SITE_URL + '/consult/" style="display:block;max-width:320px;margin:0 auto;background:#D8B13A;color:#fff;text-decoration:none;padding:16px 30px;border-radius:999px;font-weight:800;font-size:18px;">Book Your Suburb Strategy Call</a>' +
      '</div>' +
      '<p style="margin-top:20px;font-size:12px;color:#5b6770;">This quiz is guidance only. Final fit depends on inventory, timing, and your exact priorities.</p>' +
    '</div>';

  MailApp.sendEmail({
    to: toEmail,
    name: CONFIG.EMAIL_FROM_NAME,
    subject: subject,
    body: textBody,
    htmlBody: htmlBody,
    replyTo: CONFIG.ADMIN_EMAIL
  });
}

function parseQuizResultsPayload_(value) {
  if (Array.isArray(value)) {
    return value.map(normalizeQuizResultEntry_).filter(function (entry) {
      return Boolean(entry.suburb);
    });
  }

  const cleaned = clean_(value);
  if (!cleaned) {
    return [];
  }

  try {
    const parsed = JSON.parse(cleaned);
    if (!Array.isArray(parsed)) {
      return [];
    }
    return parsed.map(normalizeQuizResultEntry_).filter(function (entry) {
      return Boolean(entry.suburb);
    });
  } catch (_) {
    return [];
  }
}

function normalizeQuizResultEntry_(entry) {
  const obj = entry && typeof entry === 'object' ? entry : {};
  const suburb = clean_(obj.suburb || obj.neighborhood || obj.name);
  return {
    rank: clean_(obj.rank),
    label: clean_(obj.label) || (clean_(obj.rank) === '1' ? 'Top Pick' : 'Second Pick'),
    suburb: suburb,
    typeLabel: clean_(obj.typeLabel),
    summary: clean_(obj.summary),
    why: clean_(obj.why),
    homes: clean_(obj.homes),
    priceRange: clean_(obj.priceRange),
    homeAge: clean_(obj.homeAge),
    access: clean_(obj.access),
    amenities: clean_(obj.amenities),
    learnMoreUrl: clean_(obj.learnMoreUrl),
    image: clean_(obj.image || obj.imageUrl),
    imageAlt: clean_(obj.imageAlt || suburb),
    imageSecondary: clean_(obj.imageSecondary),
    imageSecondaryAlt: clean_(obj.imageSecondaryAlt || suburb),
    videoUrl: clean_(obj.videoUrl),
    videoTitle: clean_(obj.videoTitle)
  };
}

  const isTopPick = position === 0;
  const badgeStyle = isTopPick
    ? 'background:#2F5233;color:#fff;'
    : 'background:#dbeafe;color:#1e3a8a;';
  const videoUrl = clean_(result.videoUrl);
  const imageUrl = clean_(result.image || result.imageUrl);
  const imageSecondaryUrl = clean_(result.imageSecondary);
  const imageAlt = clean_(result.imageAlt || suburbName);
  const imageSecondaryAlt = clean_(result.imageSecondaryAlt || suburbName);
  const resolvedImageUrl = resolvePublicImageUrl_(imageUrl);
  const resolvedSecondaryImageUrl = resolvePublicImageUrl_(imageSecondaryUrl);
  const videoCaption = isTopPick

    : '';

    resolvedImageUrl,
    imageAlt,
    resolvedSecondaryImageUrl,

    ? '<div style="max-width:480px;width:100%;margin:0 auto;">' + renderClickableVideoPreview_('', videoUrl, suburbName, suburbName + ' video tour') + '</div>'

    ? '<p style="margin:0 0 14px;text-align:center;color:#4b5563;font-size:15px;line-height:1.5;">' + escapeHtml_(videoCaption) + '</p>'
    : '';
  const closingMediaHtml = isTopPick

    '<section style="margin:22px 0;padding:18px;border:1px solid #dce5ea;border-radius:14px;background:#fff;">' +
      '<div style="text-align:center;margin-bottom:8px;"><span style="' + badgeStyle + 'padding:6px 10px;border-radius:999px;font-size:12px;font-weight:700;">' + escapeHtml_(rankLabel) + '</span></div>' +

'<p style="margin:0 0 10px;">' + escapeHtml_(combineSentences_([formatPriceSentence_(result.priceRange), formatSentence_(result.homeAge), formatSentence_(result.homes)])) + '</p>' : '') +

function pickLabel_(position) {
  if (position === 0) return 'Top Pick';
  if (position === 1) return 'Second Pick';
  return 'Third Pick';
}


function buildYouTubeThumbnailUrl_(videoUrl) {
  const cleaned = clean_(videoUrl);
  if (!cleaned) {
    return '';
  }
  const match = cleaned.match(/(?:youtu\.be\/|youtube(?:-nocookie)?\.com\/(?:embed\/|watch\?v=))([A-Za-z0-9_-]{6,})/i);
  if (!match) {
    return cleaned;
  }
  return 'https://i.ytimg.com/vi/' + match[1] + '/hqdefault.jpg';
}

function normalizeYouTubeWatchUrl_(videoUrl) {
  const cleaned = clean_(videoUrl);
  if (!cleaned) {
    return '';
  }
  const match = cleaned.match(/(?:youtu\.be\/|youtube(?:-nocookie)?\.com\/(?:embed\/|watch\?v=))([A-Za-z0-9_-]{6,})/i);
  if (!match) {
    return cleaned;
  }
  return 'https://www.youtube.com/watch?v=' + match[1];
}

function resolveRenderableImageUrl_(imageUrl, videoUrl) {
  const publicImageUrl = resolvePublicImageUrl_(imageUrl);
  if (publicImageUrl) {
    return publicImageUrl;
  }
  if (clean_(videoUrl)) {
    const videoThumbnailUrl = buildYouTubeThumbnailUrl_(videoUrl);
    if (videoThumbnailUrl) {
      return videoThumbnailUrl;
    }
  }
  return '';
}

function resolvePublicImageUrl_(imageUrl) {
  const cleanedImageUrl = clean_(imageUrl);
  if (!cleanedImageUrl) {
    return '';
  }
  const publicImageUrl = resolvePublicUrl_(cleanedImageUrl);
  return /^https?:\/\//i.test(publicImageUrl) ? publicImageUrl : '';
}

function renderClickableVideoPreview_(imageUrl, videoUrl, title, altText) {
  const thumbnailUrl = resolveRenderableImageUrl_(imageUrl, videoUrl);
  const href = normalizeYouTubeWatchUrl_(videoUrl) || resolvePublicUrl_(imageUrl);
  if (!thumbnailUrl) {
    return '';
  }

  return (
    '<div style="margin:0 0 14px;border-radius:14px;overflow:hidden;border:1px solid #d7dde3;background:#fff;">' +
      '<a href="' + escapeHtml_(href) + '" target="_blank" rel="noopener noreferrer" style="display:block;text-decoration:none;">' +
        '<img src="' + escapeHtml_(thumbnailUrl) + '" alt="' + escapeHtml_(altText || title || '') + '" style="display:block;width:100%;height:auto;border:0;">' +
      '</a>' +
    '</div>'
  );
}

function combineSentences_(values) {
  return values
    .map(function (value) { return clean_(value); })
    .filter(function (value) { return Boolean(value); })
    .join(' ')
    .trim();
}

function formatPriceSentence_(value) {
  const text = clean_(value);
  if (!text) return '';
  if (/Homes here typically start/i.test(text) || /start/i.test(text) || /[.?!]$/.test(text)) {
    return text;
  }
  return 'Homes here typically start in the ' + text + ' range.';
}

function formatSentence_(value) {
  return clean_(value);
}

function resolvePublicUrl_(value) {
  const cleaned = clean_(value);
  if (!cleaned) {
    return CONFIG.BASE_SITE_URL;
  }
  if (/^https?:\/\//i.test(cleaned)) {
    return cleaned;
  }
  if (cleaned.charAt(0) === '/') {
    return CONFIG.BASE_SITE_URL + cleaned;
  }
  return cleaned;
}

  return lines.join('\n');
}

  };

function isNeighborhoodQuizPayload_(payload) {
  const haystack = [
    payload.lead_type,
    payload.quiz_type,
    payload.quiz_title,
    payload.quiz_slug,
    payload.source_key,
    payload.source_page
  ].map(function (value) { return clean_(value); }).join(' ').toLowerCase();
  return haystack.indexOf('neighborhood') !== -1;
}

function ensureQuizEmailColumns_(sheet) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(function (h) { return clean_(h); });
  [CONFIG.STATUS_COLUMN, CONFIG.SENT_AT_COLUMN, CONFIG.ERROR_COLUMN].forEach(function (name) {
    if (headers.indexOf(name) === -1) {
      sheet.insertColumnAfter(sheet.getLastColumn());
      sheet.getRange(1, sheet.getLastColumn()).setValue(name);
    }
  });
}

function markQuizEmailStatus_(sheet, rowNumber, status, sentAt, errorMessage) {
  const indexes = headerIndexes_(sheet);
  const statusCol = indexes.quiz_email_status + 1;
  const sentCol = indexes.quiz_email_sent_at + 1;
  const errorCol = indexes.quiz_email_error + 1;
  if (statusCol < 1 || sentCol < 1 || errorCol < 1) return;

  sheet.getRange(rowNumber, statusCol).setValue(clean_(status));
  sheet.getRange(rowNumber, sentCol).setValue(sentAt || '');
  sheet.getRange(rowNumber, errorCol).setValue(clean_(errorMessage));
}

function getOrCreateSheet_(ss, name) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  return sh;
}

function headerIndexes_(sheet) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const idx = function (name) { return headers.indexOf(name); };
  return {
    first_name: idx('First Name'),
    last_name: idx('Last Name'),
    email: idx('Email'),
    source_page: idx('Source Page'),
    source_key: idx('Source Key'),
    lead_type: idx('Lead Type'),
    quiz_type: idx('Quiz Type'),
    quiz_title: idx('Quiz Title'),
    quiz_city: idx('Quiz City'),
    quiz_slug: idx('Quiz Slug'),
    quiz_top_pick: idx('Quiz Top Pick'),
    quiz_second_pick: idx('Quiz Second Pick'),
    quiz_third_pick: idx('Quiz Third Pick'),
    quiz_results_json: idx('Quiz Results JSON'),
    quiz_answers_json: idx('Quiz Answers JSON'),
    quiz_email_status: idx(CONFIG.STATUS_COLUMN),
    quiz_email_sent_at: idx(CONFIG.SENT_AT_COLUMN),
    quiz_email_error: idx(CONFIG.ERROR_COLUMN)
  };
}

function clean_(v) {
  return String(v == null ? '' : v).replace(/\s+/g, ' ').trim();
}

function escapeHtml_(value) {
  return String(value == null ? '' : value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function shuffle_(arr) {
  const out = arr.slice();
  for (let i = out.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    const tmp = out[i];
    out[i] = out[j];
    out[j] = tmp;
  }
  return out;
}
