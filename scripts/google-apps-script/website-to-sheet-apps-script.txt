function doGet() {
  return json_({ ok: true, message: 'Lead endpoint is live.' });
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const raw = parseBody_(e);
    if (isTawkWebhookPayload_(raw)) {
      return handleTawkWebhook_(e, raw);
    }
    const lead = normalizeLead_(raw);
    const isReloGuideLead = isReloGuideLead_(lead);

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = getOrCreateSheet_(ss, SHEET_NAME);
    ensureHeaders_(sheet);
    const duplicateRow = lead.submission_id ? findLeadRowBySubmissionId_(sheet, lead.submission_id) : 0;
    if (duplicateRow) {
      return json_({
        ok: true,
        duplicate: true,
        duplicate_row: duplicateRow,
        email_status: 'duplicate',
        email_error: ''
      });
    }
    const rowNumber = appendLead_(sheet, lead, raw);
    const emailResult = isReloGuideLead
      ? sendReloGuideEmailAndUpdateStatus_(sheet, rowNumber, lead, raw)
      : sendLeadEmailAndUpdateStatus_(sheet, rowNumber, lead, raw);

    return json_({
      ok: true,
      email_status: emailResult.status,
      email_error: emailResult.error || ''
    });
  } catch (err) {
    return json_({
      ok: false,
      error: String(err && err.message ? err.message : err)
    });
  } finally {
    lock.releaseLock();
  }
}

function parseBody_(e) {
  const rawText = e && e.postData && e.postData.contents ? e.postData.contents : '';
  if (rawText) {
    try {
      return JSON.parse(rawText);
    } catch (_) {}
  }

  const params = (e && e.parameter) ? e.parameter : {};
  return params || {};
}

function isTawkWebhookPayload_(raw) {
  const eventName = clean_(raw && raw.event).toLowerCase();
  return eventName === 'chat:start' || eventName === 'chat:transcript_created';
}

function handleTawkWebhook_(e, raw) {
  validateTawkWebhookRequest_(e, raw);

  const eventName = clean_(raw.event).toLowerCase();
  const chatId = clean_(raw.chatId) || clean_(raw.chat && raw.chat.id);
  if (!chatId) {
    throw new Error('Tawk webhook did not include a chat ID.');
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = getOrCreateSheet_(ss, SHEET_NAME);
  ensureHeaders_(sheet);

  const normalizedPayload = normalizeTawkWebhookLead_(raw, chatId);
  const lead = normalizeLead_(normalizedPayload);
  const existingRow = findLeadRowBySubmissionId_(sheet, lead.submission_id);

  if (existingRow) {
    updateTawkLeadRow_(sheet, existingRow, lead, raw, normalizedPayload);
    return json_({
      ok: true,
      tawk_event: eventName,
      updated: true,
      row: existingRow,
      chat_id: chatId
    });
  }

  const rowNumber = appendLead_(sheet, lead, normalizedPayload);
  updateTawkLeadRow_(sheet, rowNumber, lead, raw, normalizedPayload);
  const emailResult = sendLeadEmailAndUpdateStatus_(sheet, rowNumber, lead, normalizedPayload);

  return json_({
    ok: true,
    tawk_event: eventName,
    created: true,
    row: rowNumber,
    chat_id: chatId,
    email_status: emailResult.status,
    email_error: emailResult.error || ''
  });
}

function validateTawkWebhookRequest_(e, raw) {
  const expectedToken = clean_(PropertiesService.getScriptProperties().getProperty('TAWK_WEBHOOK_TOKEN'));
  const suppliedToken = clean_(e && e.parameter && e.parameter.tawk_token);
  if (!expectedToken) {
    throw new Error('TAWK_WEBHOOK_TOKEN is not configured in Script Properties.');
  }
  if (!suppliedToken || suppliedToken !== expectedToken) {
    throw new Error('Invalid Tawk webhook token.');
  }

  const propertyId = clean_(raw && raw.property && raw.property.id);
  if (propertyId !== TAWK_PROPERTY_ID) {
    throw new Error('Unexpected Tawk property ID.');
  }
}

function normalizeTawkWebhookLead_(raw, chatId) {
  const eventName = clean_(raw.event).toLowerCase();
  const chat = raw.chat && typeof raw.chat === 'object' ? raw.chat : {};
  const visitor = eventName === 'chat:transcript_created'
    ? (chat.visitor || {})
    : (raw.visitor || {});
  const visitorName = clean_(visitor.name);
  const splitVisitorName = /^v\d+$/i.test(visitorName) ? { first: '', last: '' } : splitName_(visitorName);
  const transcript = tawkTranscriptText_(raw);
  const firstMessage = clean_(raw && raw.message && raw.message.text);
  const conversationText = transcript || firstMessage;
  const phone = tawkPhoneFromText_(conversationText);
  const domain = clean_(raw.domain);
  const sourcePage = /^https?:\/\//i.test(domain)
    ? domain
    : (domain ? 'https://' + domain.replace(/^\/+/, '') : clean_(raw.referrer));

  return {
    submitted_at: clean_(raw.time) || new Date().toISOString(),
    source_page: sourcePage,
    source_key: TAWK_SOURCE_KEY,
    lead_type: 'requested_contact',
    requested_contact_source: TAWK_SOURCE_KEY,
    requested_contact_name: visitorName,
    requested_contact_intent: 'website_live_chat',
    requested_contact_notes: conversationText,
    new_construction_message: conversationText,
    first_name: splitVisitorName.first,
    last_name: splitVisitorName.last,
    email: clean_(visitor.email),
    phone: phone,
    sms_opt_in: false,
    submission_id: 'tawk_' + chatId,
    tawk_chat_id: chatId,
    tawk_event: eventName,
    tawk_transcript: transcript,
    tawk_visitor_city: clean_(visitor.city),
    tawk_visitor_country: clean_(visitor.country),
    tawk_referrer: clean_(raw.referrer),
    tawk_property_name: clean_(raw.property && raw.property.name),
    raw_tawk_payload: raw
  };
}

function tawkTranscriptText_(raw) {
  const messages = raw && raw.chat && Array.isArray(raw.chat.messages) ? raw.chat.messages : [];
  return messages.map(function (message) {
    const text = clean_(message && message.msg);
    if (!text) return '';
    const sender = message && message.sender ? message.sender : {};
    const senderType = clean_(sender.t).toLowerCase();
    const senderLabel = senderType === 'v'
      ? 'Visitor'
      : (senderType === 'a' ? (clean_(sender.n) || 'Agent') : (clean_(sender.n) || 'System'));
    return senderLabel + ': ' + text;
  }).filter(Boolean).join('\n');
}

function tawkPhoneFromText_(text) {
  const normalized = clean_(text);
  if (!normalized) return '';
  const match = normalized.match(/(?:\+?1[\s.\-]?)?(?:\(?\d{3}\)?[\s.\-]?)\d{3}[\s.\-]?\d{4}/);
  return match ? clean_(match[0]) : '';
}

function getOrCreateSheet_(ss, name) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  return sh;
}

function ensureHeaders_(sheet) {
  removeDeprecatedColumns_(sheet);
  ensureHeadersPresent_(sheet, headers_());
}

function ensureHeadersPresent_(sheet, required) {
  const lastCol = Math.max(1, sheet.getLastColumn());
  const headerRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  const normalized = headerRow.map(function (h) { return clean_(h).toLowerCase(); });

  required.forEach(function (headerName) {
    const key = clean_(headerName).toLowerCase();
    if (normalized.indexOf(key) !== -1) {
      return;
    }

    const blankIndex = normalized.indexOf('');
    if (blankIndex !== -1) {
      sheet.getRange(1, blankIndex + 1).setValue(headerName);
      normalized[blankIndex] = key;
      return;
    }

    const appendAt = sheet.getLastColumn() + 1;
    sheet.getRange(1, appendAt).setValue(headerName);
    normalized.push(key);
  });

  if (sheet.getFrozenRows() !== 1) {
    sheet.setFrozenRows(1);
  }
}

function updateTawkLeadRow_(sheet, rowNumber, lead, raw, normalizedPayload) {
  const indexes = headerIndexes_(sheet);
  function set(headerName, value) {
    const column = indexes.byName[headerName];
    if (column < 1) return;
    sheet.getRange(rowNumber, column).setValue(value);
  }

  if (hasText_(lead.first_name)) set('First Name', lead.first_name);
  if (hasText_(lead.last_name)) set('Last Name', lead.last_name);

  set('Lead Type', 'requested_contact');
  set('Requested Contact Source', TAWK_SOURCE_KEY);
  set('Submission ID', lead.submission_id);
  set('Raw JSON', JSON.stringify(raw));
  set('Tawk Chat ID', clean_(normalizedPayload.tawk_chat_id));
  set('Tawk Event', clean_(normalizedPayload.tawk_event));

function sendLeadEmailAndUpdateStatus_(sheet, rowNumber, lead, raw) {
  try {
    if (!leadHasPhoneNumber_(lead)) {
      const skipMessage = 'No phone number provided for admin email.';
      updateLeadEmailStatus_(sheet, rowNumber, 'skipped_no_phone', '', skipMessage);
      return { status: 'skipped_no_phone', error: skipMessage };
    }
    sendLeadEmail_(lead, raw);
    updateLeadEmailStatus_(sheet, rowNumber, 'sent', new Date(), '');
    return { status: 'sent', error: '' };
  } catch (err) {
    const message = clean_(err && err.message ? err.message : err);
    updateLeadEmailStatus_(sheet, rowNumber, 'failed', '', message);
    return { status: 'failed', error: message };
  }
}

function removeDeprecatedColumns_(sheet) {
  const deprecated = {
    'name': true,
    'full phone': true,
    'requested contact name': true
  };
  const lastColumn = sheet.getLastColumn();
  if (lastColumn < 1) {
    return;
  }

  const headers = sheet.getRange(1, 1, 1, lastColumn).getValues()[0];
  const toDelete = [];
  headers.forEach(function (value, idx) {
    const normalized = clean_(value).toLowerCase();
    if (deprecated[normalized]) {
      toDelete.push(idx + 1);
    }
  });

  toDelete
    .sort(function (a, b) { return b - a; })
    .forEach(function (columnNumber) {
      sheet.deleteColumn(columnNumber);
  });
}

function leadHasPhoneNumber_(lead) {
  if (!lead || typeof lead !== 'object') {
    return false;
  }
  return hasText_(lead.phone) || hasText_(lead.home_eval_phone);
}

function updateLeadEmailStatus_(sheet, rowNumber, status, sentAt, errorMessage) {
  const indexes = headerIndexes_(sheet);
  if (indexes.email_status < 0 || indexes.email_sent_at < 0 || indexes.email_error < 0) {
    return;
  }

  let sentAtValue = '';
  if (sentAt instanceof Date && !isNaN(sentAt.getTime())) {
    sentAtValue = sentAt;
  }

  sheet
    .getRange(rowNumber, indexes.email_status + 1, 1, 3)
    .setValues([[clean_(status), sentAtValue, clean_(errorMessage)]]);
}

function sendUnsentWebsiteLeadEmails() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = getOrCreateSheet_(ss, SHEET_NAME);
  ensureHeaders_(sheet);

  const indexes = headerIndexes_(sheet);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    return {
      ok: true,
      total_rows: 0,
      pending_rows: 0,
      sent_count: 0,
      failed_count: 0
    };
  }

  const rowCount = lastRow - 1;
  const values = sheet.getRange(2, 1, rowCount, sheet.getLastColumn()).getValues();
  let pendingRows = 0;
  let sentCount = 0;
  let failedCount = 0;

  values.forEach(function (row, rowOffset) {
    const status = clean_(row[indexes.email_status]).toLowerCase();
    if (status === 'sent' || status.indexOf('skipped') === 0) {
      return;
    }

    const rowNumber = rowOffset + 2;
    const rowContext = leadContextFromSheetRow_(row, indexes);
    const isReloGuideLead = isReloGuideLead_(rowContext.lead);
    if (!isReloGuideLead && !leadHasPhoneNumber_(rowContext.lead)) {
      updateLeadEmailStatus_(sheet, rowNumber, 'skipped_no_phone', '', 'No phone number provided for admin email.');
      return;
    }
    pendingRows += 1;
    const result = isReloGuideLead
      ? sendReloGuideEmailAndUpdateStatus_(sheet, rowNumber, rowContext.lead, rowContext.raw)
      : sendLeadEmailAndUpdateStatus_(sheet, rowNumber, rowContext.lead, rowContext.raw);

    if (result.status === 'sent') {
      sentCount += 1;
    } else {
      failedCount += 1;
    }
  });

  return {
    ok: true,
    total_rows: rowCount,
    pending_rows: pendingRows,
    sent_count: sentCount,
    failed_count: failedCount
  };
}

function leadContextFromSheetRow_(row, indexes) {
  const rawJsonText = clean_(row[indexes.raw_json]) || findRawJsonInRow_(row);
  let raw = {};
  if (rawJsonText) {
    try {
      raw = JSON.parse(rawJsonText);
    } catch (_) {
      raw = {};
    }
  }
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) {
    raw = {};
  }

  const rowPayload = {
    submitted_at: toIsoString_(row[indexes.timestamp]),
    source_page: row[indexes.source_page],
    source_key: row[indexes.source_key],
    submission_id: row[indexes.submission_id],
    lead_type: row[indexes.lead_type],

    requested_contact_source: row[indexes.requested_contact_source],

function findRawJsonInRow_(row) {
  for (var i = 0; i < row.length; i++) {
    var candidate = clean_(row[i]);
    if (!candidate || candidate.charAt(0) !== '{' || candidate.charAt(candidate.length - 1) !== '}') {
      continue;
    }
    try {
      var parsed = JSON.parse(candidate);
      if (parsed && typeof parsed === 'object' && (parsed.submitted_at || parsed.source_page || parsed.email || parsed.lead_type)) {
        return candidate;
      }
    } catch (_) {}
  }
  return '';
}

function findLeadRowBySubmissionId_(sheet, submissionId) {
  const normalizedSubmissionId = clean_(submissionId);
  if (!normalizedSubmissionId) {
    return 0;
  }

  const indexes = headerIndexes_(sheet);
  if (indexes.submission_id < 0) {
    return 0;
  }

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    return 0;
  }

  const values = sheet.getRange(2, indexes.submission_id + 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < values.length; i++) {
    if (clean_(values[i][0]) === normalizedSubmissionId) {
      return i + 2;
    }
  }
  return 0;
}

function backfillMissingWebsiteLeadNames() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = getOrCreateSheet_(ss, SHEET_NAME);
  ensureHeaders_(sheet);

  const indexes = headerIndexes_(sheet);
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();

  if (lastRow < 2) {
    return {
      ok: true,
      scanned_rows: 0,
      updated_rows: 0
    };
  }

  let scannedRows = 0;
  let updatedRows = 0;

  for (let rowNumber = 2; rowNumber <= lastRow; rowNumber++) {
    const row = sheet.getRange(rowNumber, 1, 1, lastCol).getValues()[0];
    scannedRows += 1;

    const currentFirstName = clean_(row[indexes.first_name]);
    const currentLastName = clean_(row[indexes.last_name]);
    if (hasText_(currentFirstName) && hasText_(currentLastName)) {
      continue;
    }

    const raw = parseRawJsonFromRow_(row, indexes);
    const guessedName = guessLeadNameFromRaw_(raw, row, indexes);
    if (!guessedName.first && !guessedName.last) {
      continue;
    }

    const firstName = hasText_(currentFirstName) ? currentFirstName : guessedName.first;
    const lastName = hasText_(currentLastName) ? currentLastName : guessedName.last;
    if (!hasText_(firstName) && !hasText_(lastName)) {
      continue;
    }

    if (indexes.first_name >= 0 && hasText_(firstName)) {
      sheet.getRange(rowNumber, indexes.first_name + 1).setValue(firstName);
    }
    if (indexes.last_name >= 0 && hasText_(lastName)) {
      sheet.getRange(rowNumber, indexes.last_name + 1).setValue(lastName);
    }
    updatedRows += 1;
  }

  return {
    ok: true,
    scanned_rows: scannedRows,
    updated_rows: updatedRows
  };
}

function parseRawJsonFromRow_(row, indexes) {
  const rawJsonText = clean_(row[indexes.raw_json]) || findRawJsonInRow_(row);
  if (!rawJsonText) {
    return {};
  }

  try {
    const parsed = JSON.parse(rawJsonText);
    if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
      return parsed;
    }
  } catch (_) {}
  return {};
}

function guessLeadNameFromRaw_(raw, row, indexes) {
  const candidates = [
    raw && raw.first_name,
    raw && raw.firstName,
    raw && raw.requested_contact_first_name,
    raw && raw.requestedContactFirstName,
    raw && raw.home_eval_name,
    raw && raw.requested_contact_name,
    raw && raw.name,
    raw && raw.full_name
  ];

  const combinedName = splitName_(candidates.find(function (value) {
    return hasText_(value);
  }));

  let first = clean_(combinedName.first);
  let last = clean_(combinedName.last);

  if (!hasText_(first) && hasText_(row[indexes.first_name])) {
    first = clean_(row[indexes.first_name]);
  }
  if (!hasText_(last) && hasText_(row[indexes.last_name])) {
    last = clean_(row[indexes.last_name]);
  }

  if (!hasText_(first) && hasText_(raw && raw.first_name)) {
    first = clean_(raw.first_name);
  }
  if (!hasText_(last) && hasText_(raw && raw.last_name)) {
    last = clean_(raw.last_name);
  }
  if (!hasText_(first) && hasText_(raw && raw.firstName)) {
    first = clean_(raw.firstName);
  }
  if (!hasText_(last) && hasText_(raw && raw.lastName)) {
    last = clean_(raw.lastName);
  }

  if (!hasText_(first) && hasText_(raw && raw.requested_contact_name)) {
    const splitRequested = splitName_(raw.requested_contact_name);
    first = clean_(splitRequested.first);
    if (!hasText_(last)) {
      last = clean_(splitRequested.last);
    }
  }

  return {
    first: first,
    last: last
  };
}

function toIsoString_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) {
    return value.toISOString();
  }
  const parsed = new Date(value);
  if (!isNaN(parsed.getTime())) {
    return parsed.toISOString();
  }
  return new Date().toISOString();
}

function splitName_(value) {
  const normalized = clean_(value);
  if (!normalized) {
    return { first: '', last: '' };
  }
  const parts = normalized.split(' ');
  const first = clean_(parts.shift());
  const last = clean_(parts.join(' '));
  return { first: first, last: last };
}

function phoneFromFullPhone_(value) {
  const digits = String(value == null ? '' : value).replace(/[^\d]/g, '');
  if (!digits) {
    return '';
  }
  if (digits.length > 10) {
    return digits.slice(-10);
  }
  return digits;
}

function toBoolean_(value, fallback) {
  if (typeof value === 'boolean') {
    return value;
  }
  const normalized = clean_(value).toLowerCase();
  if (!normalized) {
    return Boolean(fallback);
  }
  return normalized === 'true' || normalized === '1' || normalized === 'yes' || normalized === 'y';
}

function extractQuizResults_(results) {
  if (!Array.isArray(results)) {
    return { topPick: '', secondPick: '', thirdPick: '' };
  }
  return {
    topPick: clean_(results[0] && results[0].suburb),
    secondPick: clean_(results[1] && results[1].suburb),
    thirdPick: clean_(results[2] && results[2].suburb)
  };
}

function safeJsonString_(value) {
  try {
    return JSON.stringify(value == null ? null : value);
  } catch (_) {
    return '';
  }
}

function yesNo_(value) {
  return value ? 'Yes' : 'No';
}

function normalizeRequestedContactIntent_(value) {
  const normalized = clean_(value).toLowerCase();
  if (!normalized) {
    return '';
  }
  if (
    normalized === 'looking to buy' ||
    normalized === 'looking_to_buy' ||
    normalized === 'buyer'
  ) {
    return 'looking_to_buy';
  }
  if (
    normalized === 'looking to sell' ||
    normalized === 'looking_to_sell' ||
    normalized === 'seller'
  ) {
    return 'looking_to_sell';
  }
  if (
    normalized === 'looking to sell and purchase a home' ||
    normalized === 'looking_to_sell_and_buy' ||
    normalized === 'looking to sell and buy' ||
    normalized === 'seller_and_buyer' ||
    normalized === 'sell_and_buy'
  ) {
    return 'looking_to_sell_and_buy';
  }
  return normalized.replace(/\s+/g, '_');
}

function requestedContactIntentMeta_(intent) {
  if (intent === 'looking_to_buy') {
    return { isBuyer: true, isSeller: false };
  }
  if (intent === 'looking_to_sell') {
    return { isBuyer: false, isSeller: true };
  }
  if (intent === 'looking_to_sell_and_buy') {
    return { isBuyer: true, isSeller: true };
  }
  return { isBuyer: false, isSeller: false };
}

function formatRequestedContactIntent_(intent) {
  if (intent === 'looking_to_buy') {
    return 'Looking to buy';
  }
  if (intent === 'looking_to_sell') {
    return 'Looking to sell';
  }
  if (intent === 'looking_to_sell_and_buy') {
    return 'Looking to sell and purchase a home';
  }
  return clean_(intent);
}

function hasText_(value) {
  return clean_(value) !== '';
}

function addFieldLineIfPresent_(bucket, label, value) {
  const cleaned = clean_(value);
  if (!cleaned) {
    return;
  }
  bucket.push(label + ': ' + cleaned);
}

function appendSectionIfAny_(lines, title, sectionLines) {
  if (!sectionLines || !sectionLines.length) {
    return;
  }
  lines.push('');
  lines.push(title);
  lines.push(Array(title.length + 1).join('-'));
  sectionLines.forEach(function (line) {
    lines.push(line);
  });
}

function leadTypeLabel_(leadType) {
  const normalized = clean_(leadType).toLowerCase();
  if (normalized === 'buyer') {
    return 'Buyer Lead';
  }
  if (normalized === 'seller') {
    return 'Seller Lead';
  }
  if (normalized === 'home_eval') {
    return 'Home Valuation Lead';
  }
  if (normalized === 'requested_contact') {
    return 'Requested Contact Lead';
  }
  if (normalized === 'new_construction') {
    return 'New Construction Inquiry';
  }
  return normalized ? normalized.replace(/_/g, ' ').replace(/\b\w/g, function (ch) { return ch.toUpperCase(); }) : 'Website Lead';
}

function formatDateTimeForEmail_(value) {
  const parsed = new Date(value);
  if (isNaN(parsed.getTime())) {
    return clean_(value) || new Date().toISOString();
  }
  return Utilities.formatDate(
    parsed,
    Session.getScriptTimeZone() || 'America/New_York',
    "MMM d, yyyy 'at' h:mm a z"
  );
}
